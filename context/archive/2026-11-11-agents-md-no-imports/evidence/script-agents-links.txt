#!/bin/sh
# Run from repo root. Resolves every relative markdown link of AGENTS.md (AGENTS.md sits at repo root, so GitHub resolves them relative to root).
for t in $(grep -oE '\]\([^)#]+\)' AGENTS.md | sed -E 's/^\]\((.*)\)$/\1/' | grep -vE '^[a-z]+://' | sort -u); do
  if git ls-files --error-unmatch "$t" >/dev/null 2>&1; then echo "OK tracked: $t ($(wc -c < "$t") bytes) first line: $(head -1 "$t" | cut -c1-80)"; else echo "MISSING: $t"; fi
done
echo "link lines:"; grep -nE '\]\((README\.md|docs/REFERENCE\.md)\)' AGENTS.md | cut -c1-60
